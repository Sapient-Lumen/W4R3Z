sc_proposal_close() {
  local workspace='' result='' disposition='' output='' json=0 parent leaf temporary tree candidate relative_parent
  local result_sha candidate_sha archive_tmp archive_sha receipt base base_source base_tree base_tree_sha source_manifest
  local candidate_files candidate_bytes
  while (( $# )); do
    case $1 in
      --workspace) (( $# >= 2 )) || sc_usage_die '--workspace needs a directory'; workspace=$2; shift 2 ;;
      --result) (( $# >= 2 )) || sc_usage_die '--result needs a file'; result=$2; shift 2 ;;
      --disposition) (( $# >= 2 )) || sc_usage_die '--disposition needs a value'; disposition=$2; shift 2 ;;
      --output) (( $# >= 2 )) || sc_usage_die '--output needs a file'; output=$2; shift 2 ;;
      --json) json=1; shift ;;
      *) sc_usage_die "unknown proposal close argument: $1" ;;
    esac
  done
  [[ -n $workspace && -n $result && -n $disposition && -n $output ]] \
    || sc_usage_die 'proposal close requires --workspace DIR --result FILE --disposition VALUE --output FILE'
  case $disposition in ask|recommend|hold|reject) ;; *) sc_usage_die 'disposition must be ask|recommend|hold|reject' ;; esac
  [[ -f $result && ! -L $result ]] || sc_die 'portable result must be a regular non-symlink file'
  [[ $(stat -c '%h' -- "$result") == 1 ]] || sc_die 'portable result must have one hard link'
  (( $(stat -c '%s' -- "$result") <= SC_PROPOSAL_MAX_RESULT_BYTES )) \
    || sc_die 'portable result byte bound exceeded'
  [[ ! -e $output && ! -L $output ]] || sc_die 'proposal output must be absent'
  sc_require_proposal_tools
  sc_workspace_validate "$workspace"
  parent=${output%/*}; leaf=${output##*/}; [[ $parent != "$output" ]] || parent=.
  [[ -d $parent && ! -L $parent && -n $leaf && $leaf != . && $leaf != .. ]] \
    || sc_die 'proposal output parent or leaf is unsafe'
  temporary=$(mktemp -d "$parent/.${leaf}.proposal.XXXXXXXX")
  trap "rm -rf -- $(printf '%q' "$temporary")" EXIT
  tree=$temporary/$SC_PROPOSAL_ROOT
  mkdir -m 700 -- "$tree" "$tree/source"
  candidate=$tree/CANDIDATE-MANIFEST.tsv
  sc_tree_manifest "$workspace/source" "$candidate"
  candidate_files=$SC_TREE_FILES
  candidate_bytes=$SC_TREE_BYTES
  while IFS=$'\t' read -r _digest mode _bytes relative; do
    relative_parent=${relative%/*}
    [[ $relative_parent == "$relative" ]] && relative_parent=.
    mkdir -p -- "$tree/source/$relative_parent"
    install -m "$mode" -- "$workspace/source/$relative" "$tree/source/$relative"
  done < "$candidate"
  base=$workspace/.cityb-of-chatgpt/base-source-manifest.sha256
  base_source=$temporary/base-source
  sc_source_extract_into "$base_source"
  source_manifest=$base_source/SOURCE-MANIFEST.sha256
  [[ $SC_WORKSPACE_BASE_MANIFEST_SHA == "$(sha256sum "$source_manifest" | awk '{print $1}')" ]] \
    || sc_die 'workspace donor manifest differs from the exact carrier source'
  mv -- "$source_manifest" "$temporary/embedded-source-manifest.sha256"

sc_proposal_materialize() {
  local archive='' output='' json=0 parent leaf temporary root embedded base_sha staging
  while (( $# )); do
    case $1 in
      --out) (( $# >= 2 )) || sc_usage_die '--out needs a directory'; output=$2; shift 2 ;;
      --json) json=1; shift ;;
      --*) sc_usage_die "unknown proposal materialize argument: $1" ;;
      *) [[ -z $archive ]] || sc_usage_die 'proposal materialize accepts one archive'; archive=$1; shift ;;
    esac
  done
  [[ -n $archive && -n $output ]] || sc_usage_die 'proposal materialize requires ARCHIVE --out ABSENT_DIRECTORY'
  [[ ! -e $output && ! -L $output ]] || sc_die 'proposal materialization output must be absent'
