\begin{abstract}
Several papers in this archive budget anonymity using different endpoint metrics: posterior-odds inflation (Anonymity~A), pointwise maximal leakage (PML), and maximal leakage (MaxL) in the ``Boss Fight'' repeated-use regime.
This short synthesis note pins down the exact algebraic relationships among these endpoints (under a fixed prior), and records the two sizing rules most often needed by operators and referees:
(i) MaxL is the log moment generating function of realized odds inflation; and
(ii) keeping exact-guess probability below $\alpha$ for a $K$-way secret requires total MaxL at most $\log(\alpha K)$ (nats) or $\log_2(\alpha K)$ (bits).
The goal is non-redundancy: other papers may cite this note instead of re-deriving these bridges.
\end{abstract}

\section{Setup (one secret, one observation, one prior)}
Let $S\in\mathcal S$ be a discrete secret with prior $\pi$, and let $Y\in\mathcal Y$ be an observation.
Write $P_s$ for the law of $Y$ given $S=s$ and write the public (mixture) marginal
\[
P(y)\ \defeq\ \sum_{s\in\mathcal S}\pi(s)\,P_s(y).
\]
All quantities below are properties of the channel $\{P_s\}_{s\in\mathcal S}$; only the odds-inflation / pointwise forms reference the prior through $P$.

\section{Realized odds inflation equals pointwise maximal leakage}

\begin{definition}[Realized odds inflation / pointwise Bayes factor]
For any $y$ with $P(y)>0$, define
\[
\ell(y)\ \defeq\ \max_{s\in\mathcal S}\log\frac{\pi(s\mid y)}{\pi(s)}
\ =\ \max_{s\in\mathcal S}\log\frac{P_s(y)}{P(y)}.
\]
\end{definition}

\begin{remark}[Name alignment]
Many references define \emph{pointwise maximal leakage} (PML) at outcome $y$ as
$\log\max_s \tfrac{P_s(y)}{P(y)}$ (choice of log base varies).
Thus, under a fixed prior, \emph{PML} at $y$ is exactly $\ell(y)$.
See \cite{saeidian2023pml} for operational interpretations.
\end{remark}

\section{MaxL is the log-moment of realized odds inflation}

\begin{definition}[Maximal leakage (nats) \cite{issa2020operational}]
For discrete $S,Y$, define
\[
\mathrm{MaxL}(S\to Y)\ \defeq\ \log\sum_{y\in\mathcal Y}\max_{s\in\mathcal S} P_s(y).
\]
(Using $\log_2$ gives MaxL in bits.)
\end{definition}

\begin{lemma}[Moment identity]
Let $Y\sim P$ be drawn from the mixture marginal.
Then
\[
\sum_{y\in\mathcal Y}\max_s P_s(y)\ =\ \EE_{Y\sim P}\big[\exp(\ell(Y))\big].
\]
Equivalently, in nats,
\[
\mathrm{MaxL}(S\to Y)\ =\ \log\EE_{Y\sim P}\big[\exp(\ell(Y))\big].
\]
\end{lemma}

\begin{proof}
By definition,
$\EE_P[\exp(\ell(Y))]=\sum_y P(y)\,\max_s \tfrac{P_s(y)}{P(y)}=\sum_y \max_s P_s(y)$.
Taking $\log$ gives the identity.
\end{proof}

\begin{proposition}[Two immediate corollaries]
\begin{enumerate}[leftmargin=*]
\item (Worst-case dominates.) $\mathrm{MaxL}(S\to Y)\le \sup_y \ell(y)$ (log of an average is at most the max).
\item (Typical vs worst-case.) If $\ell(Y)\le \varepsilon$ almost surely under $P$, then $\mathrm{MaxL}(S\to Y)\le \varepsilon$.
More generally, tail envelopes on $\ell(Y)$ translate into bounds on $\mathrm{MaxL}$ via the moment identity.
\end{enumerate}
\end{proposition}

\section{Sizing rules for exact identification}
The Boss-Fight series uses MaxL as a repeated-use identification budget.
The following bound is the standard ``back-of-the-envelope'' sizing rule.

\begin{lemma}[Exact-guess probability bound]
For any prior $\pi$,
\[
P_{\mathrm{guess}}(S\mid Y)\ \le\ \exp(\mathrm{MaxL}(S\to Y))\cdot \max_{s}\pi(s),
\]
where $P_{\mathrm{guess}}(S\mid Y)\defeq \EE[\max_s \pi(s\mid Y)]$.
In particular, if $S$ is uniform on $K$ secrets, then
$P_{\mathrm{guess}}(S\mid Y)\le \exp(\mathrm{MaxL})/K$.
\end{lemma}

\begin{proof}
For any outcome $y$, $\max_s \pi(s\mid y)=\max_s \pi(s)\tfrac{P_s(y)}{P(y)}\le (\max_s \pi(s))\exp(\ell(y))$.
Taking expectation under $P$ and using the moment identity yields the claim.
\end{proof}

\begin{remark}[Window budgeting]
If an operator wants $P_{\mathrm{guess}}\le \alpha$ for a uniform $K$-way secret, it suffices to enforce
\[
\mathrm{MaxL}(S\to Y_{1:T})\ \le\ \log(\alpha K)\quad\text{(nats)}
\qquad\text{or}\qquad
\mathrm{MaxL}(S\to Y_{1:T})\ \le\ \log_2(\alpha K)\quad\text{(bits)}.
\]
This is the sizing rule used in the Boss-Fight dial sheet.
\end{remark}
