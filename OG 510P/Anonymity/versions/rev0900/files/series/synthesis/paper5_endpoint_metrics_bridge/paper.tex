\begin{abstract}
This note repairs the archive's endpoint-metric crosswalk.
The ratio $\pi(s\mid y)/\pi(s)=P_s(y)/P(y)$ is posterior-\emph{mass} inflation and its worst-case logarithm is pointwise maximal leakage (PML); it is not posterior-odds inflation.
True one-vs-complement odds instead use the Bayes factor $P_s(y)/P_{-s}(y)$.
We give the exact prior-dependent conversion between the two, show by deterministic revelation that one bit of PML can coexist with infinite true-odds inflation, and preserve the valid log-moment identity linking PML to maximal leakage (MaxL) and exact-identification sizing.
All release-facing quantities are bit-valued.
\end{abstract}

\paragraph{Scope.}
Pin down the exact algebra among posterior-mass inflation, PML, MaxL, true one-vs-complement odds, and exact-guess bounds under a fixed observation channel and declared prior.

\section{Setup}
Let $S\in\mathcal S$ be a discrete secret with prior $\pi(s)>0$, and let $Y\in\mathcal Y$ be an observation.
Write $P_s$ for the law of $Y$ given $S=s$ and
\[
P(y)\defeq\sum_{s\in\mathcal S}\pi(s)P_s(y)
\]
for the mixture marginal.
Bits use $\log_2$; a natural-log source row must be divided by $\ln 2$ before reuse.

\section{Posterior-mass inflation is PML, not posterior odds}
\begin{definition}[Posterior-mass factor and PML]
For $P(y)>0$, define
\[
m_s(y)\defeq\frac{\pi(s\mid y)}{\pi(s)}=\frac{P_s(y)}{P(y)},
\qquad
\ell(y)\defeq\max_s\log_2 m_s(y).
\]
The quantity $\ell(y)$ is bit-valued pointwise maximal leakage at output $y$ under the declared prior.\cite{saeidian2023pml}
\end{definition}

The word ``mass'' is essential: $m_s(y)$ compares posterior and prior probabilities.
Odds compare a secret with an alternative.
For $p_s=\pi(s)<1$, let
\[
P_{-s}(y)\defeq\sum_{u\ne s}\frac{\pi(u)}{1-p_s}P_u(y).
\]
The one-vs-complement true-odds factor is
\[
R_s(y)\defeq
\frac{\pi(s\mid y)/(1-\pi(s\mid y))}{\pi(s)/(1-\pi(s))}
=\frac{P_s(y)}{P_{-s}(y)}.
\]
This is the familiar Bayes factor that multiplies prior odds into posterior odds; pairwise inferential frameworks likewise compare specified secret alternatives rather than a secret to the full mixture.\cite{KiferMachanavajjhala2014Pufferfish}

\begin{proposition}[Exact mass-to-odds conversion]
Let $p=\pi(s)$ and $m=m_s(y)$.
If $pm<1$, then
\[
R_s(y)=m\frac{1-p}{1-pm}.
\]
Hence a PML cap $\ell(y)\le\varepsilon$ and a prior cap $p\le p_{\max}$ imply
\[
\log_2R_s(y)
\le
\varepsilon+\log_2\frac{1-p_{\max}}{1-2^\varepsilon p_{\max}}
\]
only when $2^\varepsilon p_{\max}<1$.
\end{proposition}

\begin{proof}
Bayes gives $\pi(s\mid y)=pm$; substitution gives the identity.
For $m\ge1$, the right-hand side is increasing in both $m$ and $p$ while $pm<1$.
For $m<1$, the odds factor is below one and cannot increase the upper bound.
\end{proof}

\begin{proposition}[Finite PML does not imply finite true odds]
Let $S$ be uniform on two values and reveal $Y=S$ deterministically.
At either realized output, $m_s(y)=2$ and $\ell(y)=1$ bit, while the winning secret has posterior one and therefore infinite one-vs-complement odds inflation.
\end{proposition}

\begin{remark}[Rare-secret approximation is not an identity]
When both $p$ and $pm$ are tiny, $R_s(y)$ is close to $m_s(y)$.
A card may use that approximation only with an explicit error bound or the exact conversion above; it may not relabel PML as odds.
For the illustrative uniform prior $p=1/4096$ and $\varepsilon=0.75$ bits, the mass factor is $1.681792831$, while the guarded true-odds factor is $1.682072886$, or $0.750240221$ bits.
\end{remark}

\section{MaxL is the log-moment of PML}
\begin{definition}[Maximal leakage in bits]\label{def:maxl}
For discrete $S,Y$, define\cite{issa2020operational}
\[
\mathrm{MaxL}_2(S\to Y)\defeq\log_2\sum_{y\in\mathcal Y}\max_{s\in\mathcal S}P_s(y).
\]
\end{definition}

\begin{lemma}[Moment identity]\label{lem:moment}
For $Y\sim P$,
\[
\mathrm{MaxL}_2(S\to Y)
=
\log_2\EE_{Y\sim P}\bigl[2^{\ell(Y)}\bigr].
\]
\end{lemma}

\begin{proof}
\[
\EE_P[2^{\ell(Y)}]
=\sum_yP(y)\max_s\frac{P_s(y)}{P(y)}
=\sum_y\max_sP_s(y).
\]
Taking $\log_2$ gives the claim.
\end{proof}

\begin{proposition}[Worst-case PML dominates MaxL]
If $\ell(Y)\le\varepsilon$ almost surely under $P$, then $\mathrm{MaxL}_2(S\to Y)\le\varepsilon$.
More generally, a tail statement alone does not determine MaxL unless it controls the exponential moment in Lemma~\ref{lem:moment}.
\end{proposition}

\section{Exact-identification sizing}
\begin{lemma}[Exact-guess probability bound]\label{lem:guess}
For any prior $\pi$,
\[
P_{\mathrm{guess}}(S\mid Y)
\le
2^{\mathrm{MaxL}_2(S\to Y)}\max_s\pi(s),
\]
where $P_{\mathrm{guess}}(S\mid Y)=\EE[\max_s\pi(s\mid Y)]$.
For a uniform $K$-way secret, this becomes $P_{\mathrm{guess}}\le2^{\mathrm{MaxL}_2}/K$.
\end{lemma}

\begin{proof}
For each $y$,
$\max_s\pi(s\mid y)\le(\max_s\pi(s))2^{\ell(y)}$.
Take expectation and apply Lemma~\ref{lem:moment}.
\end{proof}

Thus a sufficient repeated-use sizing condition for a uniform $K$-way secret and target exact-guess probability $\alpha$ is
\[
\mathrm{MaxL}_2(S\to Y_{1:T})\le\log_2(\alpha K).
\]
This is an identification statement, not a true-odds statement.

\section{Release-facing validity boundary}
A downstream card must name the observation projection, prior or conditioning context, log base, repetition window, and which quantity is actually bounded:
\begin{itemize}[leftmargin=*]
\item realized PML / posterior-mass inflation $\ell(y)$;
\item channel-level MaxL;
\item a true pairwise or one-vs-complement odds factor;
\item an exact-guess cap derived from MaxL; or
\item a tail or moment row that still needs conversion.
\end{itemize}
A variable-length transcript, retry policy, state change, natural-log accountant, or altered prior creates a new claim surface unless a source-bound conversion is supplied.
In particular, neither ``PML'' nor ``Bayes factor'' may be used as a synonym for all five rows.
