\section*{Positioning inside the overlay body of work}
This note records stop-time padding facts that are additive to (and designed to be cited alongside)
\emph{Scheduling and Compiler Techniques for Metadata-Hiding Overlay Lookups}~\cite{sched} and
\emph{Spectral Anonymity and Optimal Distinguishing Bounds for Random-Walk Delegation in (Possibly Directed) Overlay Networks}~\cite{spectral}.
We therefore do \emph{not} re-derive the kernel-feasible padding model, the Kernel-TV-Pad LP, or the mixture-compiler analysis; those appear in~\cite{sched}.
Likewise, we treat~\cite{spectral} as the canonical reference for random-walk delegation anonymity; the only point of contact here is a modular lemma
that lifts stop-time indistinguishability to richer execution views.

Throughout we use the stop-time notation of~\cite{sched}:
a (secret-conditioned) success time $S\in\{1,\dots,S_{\max}\}$ is mapped to an observable stop time $T\in\{1,\dots,T_{\max}\}$ by a single upper-triangular
kernel $K$ with $K(s,t)=0$ for $t<s$.
In world $b\in\{0,1\}$, $S$ has law $\mu_b$, inducing $P_b=\mu_b K$ on $T$.
A TV-privacy budget constrains $\TV(P_0,P_1)\le \varepsilon$, and the objective is the worst-case mean
$M(K):=\max\{\E[T\mid b=0],\E[T\mid b=1]\}$.


\section{Max-mixture compilers: a baseline strictly richer than timeout-mixtures}
The mixture compiler in~\cite{sched} mixes $S$ with a secret-independent ``timeout'' draw; here we record a slightly richer, still implementable
baseline that snaps a public draw up to feasibility via a $\max$.

\begin{definition}[Max-mixture compiler]\label{def:maxmix}
Fix $\lambda\in[0,1]$ and a public distribution $Q$ on $\{1,\dots,T_{\max}\}$.
Given $S=s$, output $T=s$ with probability $\lambda$; otherwise sample $U\sim Q$ and output $T=\max\{s,U\}$.
\end{definition}

Timeout-mixtures (as analyzed in~\cite{sched}) are the special case where $Q$ is supported on $\{S_{\max},\dots,T_{\max}\}$, so $\max\{s,U\}=U$ always.

\begin{remark}[Sanity check at $S_{\max}=2$]
When $S\in\{1,2\}$, max-mixture compilers already span the global optimum of Kernel-TV-Pad: any feasible kernel can be post-processed by
$f(t)=\min\{t,2\}$ (which cannot increase TV and only decreases the mean) to an optimal kernel supported on $\{1,2\}$; but every feasible triangular
$2\times2$ kernel has the form $K(2,2)=1$ and $K(1,1)=\lambda,\ K(1,2)=1-\lambda$, i.e., a max-mixture with $Q=\delta_2$.
\end{remark}

\subsection{Minimal-horizon characterization and a certified strict separation at $S_{\max}=T_{\max}=3$}
At the minimal nontrivial horizon $S_{\max}=T_{\max}=3$, every feasible kernel can be parameterized as
\[
K=\begin{pmatrix}
 a & b & 1-a-b\\
 0 & c & 1-c\\
 0 & 0 & 1
\end{pmatrix}
\qquad (a,b,c\in[0,1],\ b\le 1-a).
\]

\begin{lemma}[Polyhedral characterization of max-mixture at $S_{\max}=T_{\max}=3$]\label{lem:maxmix3}
A feasible kernel $K$ at $S_{\max}=T_{\max}=3$ is a max-mixture compiler if and only if it satisfies the shared-tail constraint
\[
K(1,3)=K(2,3)\qquad\Longleftrightarrow\qquad c=a+b.
\]
Equivalently, the max-mixture family is the intersection of the feasible-kernel polytope with the plane $c=a+b$.
\end{lemma}

\begin{proof}
Under Definition~\ref{def:maxmix}, for $s\in\{1,2\}$ we output $3$ exactly when the public draw forces the output to $3$,
i.e., when $U=3$ and we are in the $(1-\lambda)$ branch. Hence $\Pr[T=3\mid S=1]=\Pr[T=3\mid S=2]$, which is $K(1,3)=K(2,3)$.
In the triangular parameterization this is $1-a-b=1-c$, i.e.\ $c=a+b$.
Conversely, any feasible kernel with $K(1,3)=K(2,3)$ can be implemented as a max-mixture:
take $\lambda:=K(1,1)=a$ and choose $Q$ so that $(1-\lambda)Q(3)=K(1,3)$ and $(1-\lambda)Q(2)=K(1,2)$.
\end{proof}

\begin{theorem}[Strict separation at the minimal horizon]\label{thm:sep33}
Let $\varepsilon=0.02$ and
\[
\mu_0=(0.751511627186,0.194196233939,0.054292138875),\qquad
\mu_1=(0.775286411978,0.004073598874,0.220639989148).
\]
Then the globally optimal kernel-feasible padding value is
\[
M^\star=1.592012025884,
\]
achieved by the kernel (in the above parameterization)
\[
a=0.841227383338,\qquad b=0.158772616662,\qquad c=0.125049733210,
\]
for which $\TV(P_0,P_1)=\varepsilon$.
Every max-mixture compiler satisfies the strictly worse bound
\[
M_{\mathrm{maxmix}}\ge 2.836458565568.
\]
In particular, max-mixture is strictly suboptimal already at $S_{\max}=T_{\max}=3$.
\end{theorem}

\begin{remark}[Machine-checkable certificate]
The instance in Theorem~\ref{thm:sep33} comes with explicit primal and dual witnesses (a tight primal kernel, a feasible dual, and matching objective values)
in an accompanying artifact bundle; the stop-time support remains contained in $\{1,2,3\}$ even if one enlarges $T_{\max}$, so the separation is genuinely
a \emph{minimal-horizon} phenomenon.
\end{remark}

\section{Tail-sign regime: per-row deadline normal form and a sharp extreme-point corollary}
This section records a structural reduction that does not appear in~\cite{sched}: in a broad sign regime, the TV constraint depends only on the horizon mass,
and one may restrict without loss to a low-dimensional ``per-row deadline'' family.

Let $S_{\max}=T_{\max}=K$ and define the success-time gap $\Delta(s):=\mu_0(s)-\mu_1(s)$.

\begin{definition}[Tail-sign condition (TS)]\label{def:ts}
We say $(\mu_0,\mu_1)$ satisfies \emph{tail-sign} if
\[
\Delta(1)\le 0,\ \Delta(2)\le 0,\ \dots,\ \Delta(K-1)\le 0,\qquad \Delta(K)\ge 0.
\]
\end{definition}

\begin{lemma}[Sign pattern preservation]\label{lem:tssign}
Assume (TS) and let $d:=P_0-P_1=\Delta K$ be the induced stop-time gap.
Then $d(t)\le 0$ for all $t<K$ and $d(K)\ge 0$. Consequently,
\[
\TV(P_0,P_1)=d(K).
\]
\end{lemma}

\begin{proof}[Proof sketch]
For $t<K$, $d(t)=\sum_{s\le t}\Delta(s)K(s,t)\le 0$ because (TS) gives $\Delta(s)\le 0$ for all $s\le K-1$.
Also $\sum_t d(t)=\sum_s \Delta(s)=0$, so $d(K)=-\sum_{t<K}d(t)\ge 0$.
With this sign pattern, $\|d\|_1 = \sum_{t<K}(-d(t))+d(K)=2d(K)$ and hence $\TV(P_0,P_1)=\tfrac12\|d\|_1=d(K)$.
\end{proof}


\begin{definition}[Per-row deadline kernel]\label{def:deadline}
A \emph{per-row deadline} kernel is specified by parameters $\rho_s\in[0,1]$ for $s<K$ and sets
\[
K(s,s)=\rho_s,\quad K(s,K)=1-\rho_s\ \ (s<K),\qquad K(K,K)=1,
\]
with all other entries $0$.
\end{definition}

\begin{theorem}[Per-row deadline normal form under (TS)]\label{thm:deadline}
Assume (TS). For every feasible upper-triangular kernel $K$, there exists a per-row deadline kernel $\widetilde K$ such that
\[
\TV(\mu_0\widetilde K,\mu_1\widetilde K)=\TV(\mu_0 K,\mu_1 K),
\qquad
\E_{\widetilde K}[T\mid b]\le \E_K[T\mid b]\ \text{for both}\ b=0,1.
\]
In particular, an optimum exists within the family of per-row deadline kernels.
\end{theorem}

\begin{proof}[Proof sketch]
Define $\widetilde K$ by preserving each row's ``horizon'' mass: for $s<K$, set $\widetilde K(s,K):=K(s,K)$ and put all remaining mass at the earliest feasible time $t=s$ (and $\widetilde K(K,K)=1$).
Lemma~\ref{lem:tssign} shows that under (TS) the TV constraint depends only on the last column via $d(K)=\sum_s \Delta(s)K(s,K)$, so TV is preserved.
Shifting mass from later $t\in\{s,\dots,K-1\}$ to $t=s$ (while keeping $\Pr[T=K\mid S=s]$ fixed) can only decrease the conditional mean, hence $\E_{\widetilde K}[T\mid b]\le \E_K[T\mid b]$ for both $b$.
\end{proof}


\begin{corollary}[At most two fractional rows in some optimal extreme point]\label{cor:twofrac}
Under (TS), there exists an optimal per-row deadline kernel in which at most two of the parameters $\rho_1,\dots,\rho_{K-1}$ lie strictly between $0$ and $1$.
\end{corollary}

\begin{remark}[Algorithmic consequence (pointer)]
Under Definition~\ref{def:deadline} and (TS), the optimization reduces to a small LP in $(\rho_s)$ with a single knapsack-type TV budget (Lemma~\ref{lem:tssign}).
Dualizing yields a one-parameter family of fractional knapsacks, giving an exact finite evaluation method for fixed $\varepsilon$.
\end{remark}

\section{Dual-shape sparsity beyond generic LP extreme-point sparsity}
The Kernel-TV-Pad LP in~\cite{sched} already implies \emph{generic} sparsity (few nonzeros overall).
The following refinement gives a \emph{row-wise} support bound when one has a dual test function with few sign switches.

\begin{lemma}[Few-switch duals imply few candidate minimizers]\label{lem:fews}
In the standard TV-linearized dual for Kernel-TV-Pad, suppose the normalized dual test function
$\sigma:\{1,\dots,T_{\max}\}\to[-1,1]$ is piecewise constant with at most $k$ switch points.
Fix a row index $s$; the complementary-slackness ``argmin-support'' rule implies that row $s$ may place mass only on minimizers of an affine score
$C_s(t)=t q_s + \Delta(s)\cdot w\,\sigma(t)$ over $t\ge s$, where $q_s>0$ and $w\ge 0$ are dual weights.
Then every minimizer is attained at the earliest time of one of the constant segments intersecting $\{s,\dots,T_{\max}\}$,
so the minimizer set has size at most $k+1$.
\end{lemma}

\begin{corollary}[Row-support bound]\label{cor:rowsupp}
If an optimal dual witness admits a $k$-switch normalized test function $\sigma$, then there exists an optimal primal kernel whose
every row is supported on at most $k+1$ stop times.
\end{corollary}

\section{Overlay bridge lemma: lifting stop-time privacy to richer execution views}
To connect stop-time padding contracts to overlay executions (as in the compiler-calculus view of~\cite{sched} and the delegation analyses in~\cite{spectral}),
it is useful to lift a TV guarantee on $T$ to a TV guarantee on a richer view $V$.

\begin{lemma}[View privacy from stop-time privacy under a failure event]\label{lem:bridge}
For each world $b\in\{0,1\}$ let $T_b$ be a stop time and let $V_b$ be an execution view.
Assume there exists an event $E_b$ such that, conditioned on $E_b$, the view is (up to negligible statistical distance) a post-processing of the stop time:
\[
\TV\bigl( V_b\mid E_b,\ \Phi(T_b)\bigr)\le \mathrm{negl},
\]
for some randomized mapping $\Phi$ independent of $b$. Then
\[
\TV(V_0,V_1)\ \le\ \Prb[\neg E_0]+\Prb[\neg E_1]+\TV(T_0,T_1)+\mathrm{negl}.
\]
\end{lemma}

\begin{remark}
Lemma~\ref{lem:bridge} isolates a common proof pattern: establish $\TV(T_0,T_1)\le \varepsilon$ by Kernel-TV-Pad,
then identify an event $E_b$ under which the attacker view factors (approximately) through $T_b$, and separately bound $\Pr[\neg E_b]$.
\end{remark}
