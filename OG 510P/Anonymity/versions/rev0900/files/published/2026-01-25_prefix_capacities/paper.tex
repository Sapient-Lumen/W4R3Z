\begin{abstract}
This note is a compact addendum to the termination-time (stop-time) channel in
\emph{Scheduling and Compiler Techniques for Metadata-Hiding Overlay Lookups}~\cite{schedcompiler}.
The main paper optimizes over padding kernels (Kernel-TV-Pad)~\cite{schedcompiler}.
Here we instead characterize useful \emph{projections} of that kernel polytope---deadlines and average stop-time laws---via closed forms and reusable cuts.

We provide three mechanically reusable facts.
(1) \textbf{Prefix capacities:} for every deadline $t$, a sharp closed-form characterization of the maximal achievable probability of stopping by $t$
under a TV budget, computable by a greedy fractional-knapsack routine.
(2) \textbf{Not sufficient:} tight per-deadline bounds do not characterize global feasibility; we give a minimal rational counterexample at horizon $T=3$.
(3) \textbf{Membership \& separation:} an LP $\TV_{\min}(Q)$ certifies whether a proposed average stop-time law $Q$ is implementable at a given budget and returns a violated linear inequality when it is not.
We omit large artifacts and give reconstruction-oriented specifications.

\smallskip
\noindent\textbf{Optional library.} Appendices record additional structure mined from earlier drafts: a transport interpretation for the worst-case objective and two explicit $T=3$ ``library facts.''
\end{abstract}

\section{Context, use-cases, and model}\label{sec:model}

\paragraph{When to use this note (instead of just running Kernel-TV-Pad).}
Kernel-TV-Pad~\cite{schedcompiler} is the right tool when you want a numeric optimum for a concrete instance.
The results here are useful when you want \emph{fast, explainable structure} in the lower-dimensional space of stop-time laws:
(i) quick deadline bounds and sanity checks,
(ii) a cheap tuning surface for choosing deadlines/budgets before running a full LP,
(iii) a cutting-plane workflow over averages $Q$ via membership/separation cuts (Section~\ref{sec:membership}),
and (iv) a lemma-pack that an automated reader can reuse without re-deriving the full kernel LP.

\paragraph{Discrete success times and shared delay-only kernels.}
Fix a discrete horizon $\{1,\dots,T\}$.
A success time $S\in\{1,\dots,T\}$ is drawn from one of two laws $\mu_0,\mu_1$ (``worlds'' / ``secrets'').
A padding rule is a \emph{shared delay-only kernel} $K$:
an upper-triangular row-stochastic matrix with entries $K(s,t)=\Prb[Y=t\mid S=s]$ for $t\ge s$ and $K(s,t)=0$ for $t<s$.
The induced stop-time laws are
\[
P_b(t) = \sum_{s\le t} \mu_b(s)\,K(s,t),\qquad b\in\{0,1\}.
\]
We also use the average stop-time law
\[
Q := \tfrac12(P_0+P_1).
\]

\paragraph{Privacy budget (TV).}
If the adversary observes only $Y$, the optimal distinguishing advantage is
\[
\TV(P_0,P_1)=\tfrac12\sum_{t=1}^T |P_0(t)-P_1(t)|.
\]
We enforce approximate hiding by $\TV(P_0,P_1)\le \Delta$.

\paragraph{Mass and charge.}
Define the average \emph{mass} and signed \emph{charge} on success times:
\[
m(s) := \tfrac12(\mu_0(s)+\mu_1(s)),\qquad
w(s) := \mu_0(s)-\mu_1(s),
\]
so $\sum_s m(s)=1$ and $\sum_s w(s)=0$.

\section{Prefix capacities: sharp bounds for deadlines}\label{sec:capacity}

Fix a deadline $t<T$ and write the average prefix CDF $F_Q(t):=\Prb[Y\le t]_{\mathrm{avg}}=\sum_{u\le t} Q(u)$.

\subsection{The prefix-capacity LP}
Only rows $s\le t$ can contribute to the event $\{Y\le t\}$ under a delay-only kernel.
For $t<T$, define the \emph{prefix capacity}
\begin{equation}\label{eq:cap}
\mathrm{cap}_t(\Delta)
:=
\max\Big\{\sum_{s\le t}m(s)y_s:\ 0\le y_s\le 1,\ \big|\sum_{s\le t}w(s)y_s\big|\le \Delta\Big\}.
\end{equation}

\subsection{Sharp deadline theorem}
\begin{theorem}[Prefix capacity is tight]\label{thm:cap-tight}
For every $t<T$,
\[
\max_{\substack{K\ \text{shared delay-only}\\ \TV(P_0,P_1)\le \Delta}} F_Q(t)\ =\ \mathrm{cap}_t(\Delta).
\]
\end{theorem}

\begin{proof}
Fix a shared delay-only kernel $K$ and define, for each $s\le t$,
$y_s := \Prb[Y\le t\mid S=s]=\sum_{u\le t}K(s,u)$.
Then $0\le y_s\le 1$ and
\[
F_Q(t)=\sum_{s\le t}m(s)y_s,\qquad
F_0(t)-F_1(t)=\sum_{s\le t}w(s)y_s.
\]
Since $\{1,\dots,t\}$ is an event, $|F_0(t)-F_1(t)|\le \TV(P_0,P_1)\le \Delta$, so $(y_s)$ is feasible for~\eqref{eq:cap} and $F_Q(t)\le \mathrm{cap}_t(\Delta)$.

Conversely, given any feasible $(y_s)_{s\le t}$, define a kernel that outputs only at times $t$ and $T$:
for $s\le t$, set $K(s,t)=y_s$ and $K(s,T)=1-y_s$; for $s>t$, set $K(s,T)=1$.
Then $F_Q(t)=\sum_{s\le t}m(s)y_s$ and
\[
\TV(P_0,P_1)=|F_0(t)-F_1(t)|=\big|\sum_{s\le t}w(s)y_s\big|\le \Delta,
\]
so the bound is achievable.
\end{proof}

\subsection{Greedy computation (fractional knapsack)}
If $\big|\sum_{s\le t} w(s)\big|\le \Delta$, then $\mathrm{cap}_t(\Delta)=\sum_{s\le t} m(s)$ (take $y_s=1$).
Otherwise, to reduce the prefix charge into $[-\Delta,\Delta]$, decrease some $y_s$ from $1$.
Decreasing $y_s$ by $\alpha$ removes mass $\alpha m(s)$ and removes charge $\alpha w(s)$, so the mass cost per unit removed charge is $m(s)/|w(s)|$.
Thus the optimum is obtained by removing charge starting from indices with largest $|w(s)|/m(s)$ and stopping fractionally on the last index.

\paragraph{Pseudocode (one deadline).} Handle $w(s)=0$ by treating $|w(s)|/m(s)=0$.
\begin{verbatim}
Input: arrays m[1..t], w[1..t], budget Delta
Initialize y[s] = 1 for all s<=t
c = sum_{s<=t} w[s]
if |c| <= Delta: return cap = sum_{s<=t} m[s], y

need = |c| - Delta
Sort indices s<=t by decreasing r[s] = |w[s]| / m[s]
for s in sorted order:
    if w[s] == 0: continue
    removable = |w[s]|
    take = min(need, removable)
    y[s] = 1 - take / removable
    need -= take
    if need == 0: break

cap = sum_{s<=t} m[s] * y[s]
return cap, y
\end{verbatim}

\section{Tight prefix bounds are not sufficient}\label{sec:envelope}

Theorem~\ref{thm:cap-tight} yields necessary inequalities for any achievable average law $Q$:
\begin{equation}\label{eq:prefix-ineq}
F_Q(t) \le \mathrm{cap}_t(\Delta)\qquad \forall t<T.
\end{equation}
A tempting relaxation saturates each prefix and takes differences:
\[
F_{\mathrm{env}}(t):=\mathrm{cap}_t(\Delta),\quad
Q_{\mathrm{env}}(t):=F_{\mathrm{env}}(t)-F_{\mathrm{env}}(t-1),\quad F_{\mathrm{env}}(0):=0.
\]
This envelope respects every inequality~\eqref{eq:prefix-ineq} but can be globally infeasible because a single shared kernel must satisfy \emph{all} deadlines at once.

\begin{proposition}[Minimal envelope gap at $T=3$]\label{prop:T3gap}
Let $T=3$, $\Delta=\tfrac{1}{20}$, and
\[
\mu_0=\Big(\tfrac{19}{60},\tfrac{2}{5},\tfrac{17}{60}\Big),\qquad
\mu_1=\Big(\tfrac{4}{15},\tfrac{1}{2},\tfrac{7}{30}\Big).
\]
Then $\mathrm{cap}_1(\Delta)=\tfrac{7}{24}$ and $\mathrm{cap}_2(\Delta)=\tfrac{89}{120}$, so
\[
Q_{\mathrm{env}}=\Big(\tfrac{7}{24},\tfrac{9}{20},\tfrac{31}{120}\Big).
\]
Yet $Q_{\mathrm{env}}$ is infeasible at budget $\Delta$: its minimum required TV satisfies $\TV_{\min}(Q_{\mathrm{env}})=\tfrac{1}{10}>\Delta$.
Equivalently, $Q_{\mathrm{env}}$ violates the valid inequality
\[
\frac{124}{315}Q(1)+\frac{2}{9}Q(2)\ \le\ \frac{89}{540},
\quad\text{i.e.}\quad 1488\,Q(1)+840\,Q(2)\le 623.
\]
(The inequality is a dual certificate returned by $\TV_{\min}$ in Section~\ref{sec:membership}.)
\end{proposition}

\section{\texorpdfstring{Membership and separation via $\TV_{\min}(Q)$}{Membership and separation via TVmin(Q)}}\label{sec:membership}

Let $\cQ_\Delta\subseteq\mathbb{R}^T$ be the set of all achievable average stop-time laws $Q$ under shared delay-only kernels with $\TV(P_0,P_1)\le \Delta$.
Prefix inequalities are necessary but not sufficient (\Cref{prop:T3gap}). This section gives a compact LP that decides membership and returns separating cuts.

\subsection{A membership LP}
For a candidate average law $Q^\circ$, define
\[
\TV_{\min}(Q^\circ)
:=\min\bigl\{\TV(P_0,P_1):\ \exists\ \text{shared delay-only }K\text{ with } \tfrac12(\mu_0K+\mu_1K)=Q^\circ\bigr\}.
\]
This is a linear program.

\paragraph{LP specification (mechanically implementable).}
Introduce variables
$K(s,t)\ge 0$ $(1\le s\le t\le T)$,
$P_0(t),P_1(t)\ge 0$,
and $u(t)\ge 0$ for $t=1,\dots,T$,
with constraints:
\begin{align*}
&\text{(row sums)} &&\sum_{t=s}^T K(s,t)=1\qquad \forall s,\\
&\text{(pushforward)} &&P_b(t)=\sum_{s\le t}\mu_b(s)\,K(s,t)\qquad \forall t,\ b\in\{0,1\},\\
&\text{(match target)} &&\tfrac12(P_0(t)+P_1(t))=Q^\circ(t)\qquad \forall t,\\
&\text{(linearize TV)} &&u(t)\ge P_0(t)-P_1(t),\quad u(t)\ge -(P_0(t)-P_1(t))\qquad \forall t.
\end{align*}
Minimize $\tfrac12\sum_{t=1}^T u(t)$. The optimum equals $\TV_{\min}(Q^\circ)$.

\begin{theorem}[Membership and separation]\label{thm:memsep}
Fix $\Delta$.
\begin{enumerate}[leftmargin=1.5em]
\item \textbf{Membership:} $Q^\circ\in\cQ_\Delta$ if and only if $\TV_{\min}(Q^\circ)\le \Delta$.
\item \textbf{Separation:} If $Q^\circ\notin\cQ_\Delta$, LP duality returns an explicit vector $g\in\mathbb{R}^T$ such that
$g\cdot Q\le \sup\{g\cdot Q':Q'\in\cQ_\Delta\}$ holds for all $Q\in\cQ_\Delta$ but is violated by $Q^\circ$.
\end{enumerate}
\end{theorem}

\begin{remark}[Cut extraction in one paragraph]\label{rem:cut}
In practice, one solves the primal LP above and reads the dual multipliers on the ``match target'' constraints
$\tfrac12(P_0(t)+P_1(t))=Q^\circ(t)$.
Those multipliers define the separating direction $g$ in \Cref{thm:memsep}.
If $\TV_{\min}(Q^\circ)>\Delta$, the dual optimal solution \emph{is} a certificate: it yields a concrete linear inequality in $Q$
that is valid for $\cQ_\Delta$ and violated by $Q^\circ$ (as in \Cref{prop:T3gap}).
\end{remark}

\section{Reconstruction specification (short)}\label{sec:recon}

We omit large artifacts; the results are intended to be reconstructible from the specification:
\begin{enumerate}[leftmargin=1.5em, itemsep=2pt]
\item Compute $m,w$ from $(\mu_0,\mu_1)$.
\item For each deadline $t$, compute $\mathrm{cap}_t(\Delta)$ by the greedy fractional-knapsack routine.
\item Decide feasibility of any candidate $Q^\circ$ at budget $\Delta$ by solving $\TV_{\min}(Q^\circ)$ and, if infeasible, extracting a dual cut.
\end{enumerate}

\section{Discussion}
Prefix capacities give sharp per-deadline bounds for shared-kernel termination-time hiding.
The envelope counterexample shows global feasibility is genuinely polyhedral: sharp local bounds do not characterize the projected feasible set.
Finally, $\TV_{\min}$ provides a practical bridge between full kernel optimization (Kernel-TV-Pad~\cite{schedcompiler}) and coarse relaxations:
it certifies designs and produces explicit separating cuts.

\section{Optional: transport equivalence for the worst-case objective}\label{app:transport}

This appendix records additional structure for the worst-case overhead objective
$\E_{\max}[Y]=\max\{\E[Y\mid 0],\E[Y\mid 1]\}$.
It is not needed for the main story (Sections~\ref{sec:capacity}--\ref{sec:membership}).

\paragraph{Shared-kernel minimax optimization (KI).}
For a budget $\tau\in[0,\tau_{\max}]$ where $\tau_{\max}:=\TV(\mu_0,\mu_1)$, define
\[
\mathrm{OPT}_{KI}(\tau):=\min_{K\ \text{shared delay-only}}\ \max_{b\in\{0,1\}}\ \E[Y\mid b]\quad\text{s.t.}\quad \TV(\mu_0K,\mu_1K)\le \tau.
\]
Let $\tau_{\min}:=\min_K \TV(\mu_0K,\mu_1K)$ be the minimum achievable leakage.

\paragraph{Unbalanced transport (UT-$\tau$).}
Let $d:=\mu_0-\mu_1$ and define $P:=\{i:d_i>0\}$, $N:=\{j:d_j<0\}$ with $p_i:=d_i$ for $i\in P$ and $q_j:=-d_j$ for $j\in N$.
A UT plan has variables $f_{ij}\ge 0$ ($i\in P,j\in N$) and unmatched masses $u_i\ge 0$ ($i\in P$), $v_j\ge 0$ ($j\in N$) such that
\[
\sum_{j\in N} f_{ij} + u_i = p_i,\qquad
\sum_{i\in P} f_{ij} + v_j = q_j,
\]
and unmatched mass $r:=\sum_{i\in P}u_i=\sum_{j\in N}v_j$ satisfies $r\le \tau$.
Let $m(i,j):=\max\{i,j\}$ and scaling factors $A_i^{(b)}:=\mu_b(i)/p_i$ ($i\in P$) and $B_j^{(b)}:=\mu_b(j)/q_j$ ($j\in N$).
Define
\[
E_b(f,u,v):=\sum_{i\in P,\ j\in N} (A_i^{(b)}+B_j^{(b)})\,m(i,j)\,f_{ij}
+\sum_{i\in P} A_i^{(b)}\, i\, u_i
+\sum_{j\in N} B_j^{(b)}\, j\, v_j
+\sum_{s:d_s=0} \mu_b(s)\, s.
\]
Set $\mathrm{OPT}_{UT}(\tau):=\min_{(f,u,v):\,r\le \tau}\ \max\{E_0(f,u,v),E_1(f,u,v)\}$.

\begin{theorem}[KI $\Leftrightarrow$ UT (exact leakage)]\label{thm:ki-ut}
For all $\tau\in[0,\tau_{\max}]$ (and taking $\mathrm{OPT}_{KI}(\tau)=\mathrm{OPT}_{UT}(\tau)=+\infty$ for $\tau<\tau_{\min}$),
\[
\mathrm{OPT}_{KI}(\tau)=\mathrm{OPT}_{UT}(\tau).
\]
\end{theorem}

\begin{proposition}[Value function properties]\label{prop:value}
$\mathrm{OPT}_{KI}(\tau)$ is finite on $[\tau_{\min},\tau_{\max}]$, equals $+\infty$ on $[0,\tau_{\min})$, and is nonincreasing, convex,
and piecewise-linear on $[\tau_{\min},\tau_{\max}]$.
\end{proposition}

\section{Optional: two $T=3$ library facts}\label{app:small}

\subsection{Perfect hiding at $T=3$: closed-form feasible averages}
Let $T=3$ and write $\mu_0=(a_1,a_2,a_3)$, $\mu_1=(b_1,b_2,b_3)$.
Let $\cQ_{KI}$ denote the set of average stop-time laws $Q=\tfrac12(\mu_0K+\mu_1K)$ achievable with perfect hiding ($\mu_0K=\mu_1K$).

\begin{theorem}[Closed form for $\cQ_{KI}$ when $T=3$]\label{thm:qki-n3}
Assume $\mu_0\neq \mu_1$.
\begin{enumerate}[leftmargin=1.5em]
\item (\textbf{Triangle case:} $a_1=b_1$.) Then $\cQ_{KI}$ is exactly
\[
Q(1)\ge 0,\quad Q(2)\ge 0,\quad Q(1)+Q(2)\le a_1,\quad Q(3)=1-Q(1)-Q(2).
\]
\item (\textbf{Edge case:} $a_1\neq b_1$.) Then every $Q\in\cQ_{KI}$ satisfies $Q(1)=0$, so $\cQ_{KI}=\{(0,q,1-q): q\in[0,q_{\max}]\}$, where:
if $a_2=b_2$, then $q_{\max}=a_2$; otherwise let $r:=-(a_1-b_1)/(a_2-b_2)$.
If $r\le 0$, then $q_{\max}=0$; if $r>0$, then $q_{\max}=\min\{a_1+a_2 r,\ a_2+a_1/r\}$.
\end{enumerate}
\end{theorem}

\subsection{Intermediate leakage: an explicit strict chord gap}
\begin{proposition}[Strict chord gap at $T=3$]\label{prop:mixture-gap}
Let $T=3$ and $\mu_0=(4,3,1)/8$, $\mu_1=(1,2,5)/8$.
Then $\TV(\mu_0,\mu_1)=1/2$ and perfect hiding is feasible with an optimal perfect-hiding target $Q_*^{KI}=\delta_3$.
At budget $\tau=1/4$, the endpoint-mixture baseline yields worst-case mean $z_{\mathrm{mix}}=11/4$.
However, the shared kernel
\[
K(1,1)=1/3,\ K(1,3)=2/3;\quad K(2,2)=1;\quad K(3,3)=1
\]
induces $\TV(P_0,P_1)=1/4$ and achieves $\max\{\E[P_0],\E[P_1]\}=\E[P_1]=8/3<11/4$.
\end{proposition}
